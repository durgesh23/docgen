% Patch the \LT@array macro to modify column specifications
\makeatletter
\patchcmd{\LT@array}{\@mkpream{#2}}{
  \StrGobbleLeft{#2}{2}[	empPream] % Remove @{} from the start
  \StrGobbleRight{\tempPream}{2}[	empPream] % Remove @{} from the end
  \StrSubstitute{\tempPream}{l}{|l}[	empPream] % Substitute l with |l
  \StrSubstitute{\tempPream}{c}{|c}[	empPream] % Substitute c with |c
  \StrSubstitute{\tempPream}{r}{|r}[	empPream] % Substitute r with |r
  \edef\finalpream{@{}\tempPream|@{}} % Add | at the end and surround by @{}
  \@mkpream{\finalpream} % Pass the final preamble to \@mkpream
}{}{}
\makeatother
